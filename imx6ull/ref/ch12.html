12 · 驱动开发入门

应用层调 write()，内核里由驱动去写寄存器。驱动就是这两者之间的翻译官。这一章从「加载一个模块」走到「应用能控制一个 LED」。

本章目标

① 编出并加载第一个内核模块；② 理解主次设备号与 file_operations；③ 写一个字符设备驱动并自动创建 /dev 节点；④ 在驱动里操作 GPIO；⑤ 用设备树 compatible 匹配自己的驱动。

12.1 驱动在哪一层

应用程序open / read / write
↓
设备文件/dev/myled
↓
字符设备驱动file_operations
↓
GPIO 子系统gpiod / pinctrl
↓
寄存器IOMUXC / GPIO_DR

应用永远不知道寄存器地址，它只认 /dev/myled 这个文件。驱动负责把「写一个字符」翻译成「把某个引脚拉高」。

12.2 第一个内核模块

// hello_mod.c
#include <linux/module.h>
#include <linux/init.h>

static int __init hello_init(void)
{
    printk(KERN_INFO "hello_mod: loaded\n");
    return 0;
}

static void __exit hello_exit(void)
{
    printk(KERN_INFO "hello_mod: removed\n");
}

module_init(hello_init);
module_exit(hello_exit);
MODULE_LICENSE("GPL");
MODULE_AUTHOR("you");
MODULE_DESCRIPTION("first module");

Makefile（注意内核源码必须先编译过一次，否则没有 Module.symvers）：

obj-m := hello_mod.o
KDIR  := /home/ubuntu/linux-imx
PWD   := $(shell pwd)

all:
	$(MAKE) -C $(KDIR) M=$(PWD) \
		ARCH=arm CROSS_COMPILE=arm-linux-gnueabihf- modules

clean:
	$(MAKE) -C $(KDIR) M=$(PWD) clean
$ make
$ scp hello_mod.ko root@192.168.1.101:/root/
# 板子上
# insmod hello_mod.ko
# dmesg | tail -3
# lsmod | grep hello
# rmmod hello_mod
printk 不是 printf 它写进内核环形缓冲区，用 dmesg 看，不输出到你的终端。日志级别 KERN_INFO、KERN_ERR 决定它会不会被默认级别过滤掉。

12.3 字符设备：从设备号到 /dev 节点

概念含义
主设备号标识「哪一类驱动」，如 240
次设备号标识「同一驱动下的第几个设备」
file_operations函数指针表：open/read/write/ioctl 都在这注册
cdev内核里代表一个字符设备的结构体
class_create在 /sys/class 下建类，为自动创建设备节点做准备

老接口 register_chrdev 一行搞定但只给一个主设备号、无法精细控制；新接口是 alloc_chrdev_region + cdev_init + cdev_add，工程上只用新接口。

12.4 一个能用的字符设备驱动（骨架）

#include <linux/module.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/device.h>
#include <linux/uaccess.h>

#define DEV_NAME  "mychar"
#define BUF_SIZE  64

static dev_t        dev_id;
static struct cdev  my_cdev;
static struct class *my_class;
static char         kbuf[BUF_SIZE];

static int my_open(struct inode *in, struct file *fp)
{
    printk(KERN_INFO "mychar: open\n");
    return 0;
}
static int my_release(struct inode *in, struct file *fp) { return 0; }

static ssize_t my_read(struct file *fp, char __user *ubuf,
                       size_t len, loff_t *off)
{
    int ret = simple_read_from_buffer(ubuf, len, off, kbuf, BUF_SIZE);
    return ret;
}

static ssize_t my_write(struct file *fp, const char __user *ubuf,
                        size_t len, loff_t *off)
{
    if (len > BUF_SIZE) len = BUF_SIZE;
    if (copy_from_user(kbuf, ubuf, len)) return -EFAULT;
    printk(KERN_INFO "mychar: got %zu bytes\n", len);
    return len;
}

static const struct file_operations my_fops = {
    .owner   = THIS_MODULE,
    .open    = my_open,
    .release = my_release,
    .read    = my_read,
    .write   = my_write,
};

static int __init my_init(void)
{
    alloc_chrdev_region(&dev_id, 0, 1, DEV_NAME);
    cdev_init(&my_cdev, &my_fops);
    cdev_add(&my_cdev, dev_id, 1);

    my_class = class_create(THIS_MODULE, DEV_NAME);
    device_create(my_class, NULL, dev_id, NULL, DEV_NAME);

    printk(KERN_INFO "mychar: major=%d minor=%d\n",
           MAJOR(dev_id), MINOR(dev_id));
    return 0;
}

static void __exit my_exit(void)
{
    device_destroy(my_class, dev_id);
    class_destroy(my_class);
    cdev_del(&my_cdev);
    unregister_chrdev_region(dev_id, 1);
}

module_init(my_init);
module_exit(my_exit);
MODULE_LICENSE("GPL");
# 板子上
# insmod mychar.ko
# dmesg | tail -2          # 记下 major
# ls -l /dev/mychar        # 应该自动出现，c 开头
# echo "hello" > /dev/mychar
# cat /dev/mychar
# dmesg | tail -2
节点没自动出现？ ① 有没有 class_create + device_create ② 板子上 udev/mdev 有没有跑（BusyBox 环境要 mdev -s）③ 手动救急：mknod /dev/mychar c <major> 0。

12.5 在驱动里操作 GPIO

#include <linux/gpio.h>
#include <linux/of_gpio.h>

static int led_gpio;

/* probe 里：从设备树取引脚号 */
led_gpio = of_get_named_gpio(dev->of_node, "led-gpios", 0);
if (led_gpio < 0) return led_gpio;

gpio_request(led_gpio, "myled");
gpio_direction_output(led_gpio, 0);

/* 控制 */
gpio_set_value(led_gpio, 1);   /* 高电平 */
gpio_set_value(led_gpio, 0);

/* remove 里 */
gpio_free(led_gpio);

新内核推荐 gpiod 描述符接口（devm_gpiod_get / gpiod_set_value），能自动释放资源、自动处理有效电平，写产品代码时用它。

gpio_request 失败 = 引脚被别人占了 dmesg 里会打印 gpio ... already requested。常见原因：设备树里 gpio-leds 节点已经把同一个引脚占用了。一个引脚同时只能有一个使用者，写自己的驱动前先把原生 LED 节点 status = "disabled"。

12.6 用设备树匹配驱动

static const struct of_device_id my_of_match[] = {
    { .compatible = "myvendor,myled" },
    { /* sentinel */ }
};
MODULE_DEVICE_TABLE(of, my_of_match);

static struct platform_driver my_driver = {
    .probe  = my_probe,
    .remove = my_remove,
    .driver = {
        .name           = "myled",
        .of_match_table = my_of_match,
    },
};
module_platform_driver(my_driver);

设备树里对应写：

myled {
    compatible = "myvendor,myled";
    led-gpios = <&gpio1 3 GPIO_ACTIVE_HIGH>;
    pinctrl-names = "default";
    pinctrl-0 = <&pinctrl_led>;
    status = "okay";
};

compatible 字符串必须逐字符一致，包括厂商前缀的大小写。不一致，probe 根本不会被调用，lsmod 看着加载成功但什么也没发生——这是最常见的「驱动加载了没反应」。

12.7 完整流程串一遍

  1. 改设备树，加 myled 节点与 pinctrl，make dtbs。
  2. 编驱动 .ko，传到板子。
  3. insmod myled.ko，dmesg 看 probe 有没有执行。
  4. ls /dev/myled，不存在就 mknod 或检查 class 创建。
  5. 应用程序 open("/dev/myled") → write("1") → LED 亮。
  6. 看 /sys/class/gpio 或万用表验证电平。

12.8 动手任务

  • 编 hello_mod 模块，加载、查看 dmesg、卸载，全程记录输出。
  • 编字符设备驱动，实现 write 存数据、read 读回来，用 echo / cat 验证。
  • 把 GPIO 加进驱动，让 write "1"/"0" 控制板子上的 LED。
  • 在设备树里改 compatible 为你的名字，验证 probe 被调用；再故意改成别的字符串，观察 probe 不再执行。
  • 让另一个平台驱动同时请求同一个 GPIO，记录 gpio_request 失败时的报错。
  • 把驱动写成开机自动加载（/etc/modules 或 modprobe 配置）。

12.9 验收标准

  • 能独立编出 .ko 并加载/卸载
  • 能写字符设备驱动，应用可读写，/dev 节点自动出现
  • 能解释主次设备号、file_operations、cdev 的作用
  • 能在驱动里申请并使用一个 GPIO 控制 LED
  • 能用 compatible 匹配平台驱动，并知道不匹配时的现象
  • 遇到「加载成功但没反应」能按 probe → dts → 引脚占用三步排查
