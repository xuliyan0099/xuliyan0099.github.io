    ['111', '无', '无', '无', '无', '任意其它值']

    var y = 236 + i * 52, hi = i === 0 ? 'var(--alert-line)' : 'var(--line)';

    box(1632, y, 236, 44, 'var(--page)', 'var(--line)', 2);
