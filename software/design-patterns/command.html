命令模式COMMAND

把“要做什么”变成一个可传递、可排队、可回滚的对象。调用者只捏着命令指针，不知道它改的是增益还是模式；撤销也不是“反向再算一遍”，而是命令在 exec 时记下旧值，undo 时照着还原。

01问题：直接调用，就没有“以后”和“回退”

最省事的写法是直接 dev.gain = 20;。它有三个躲不开的后果：

  • 没有队列。想等总线空闲再写、想合并同一参数的多次设置，都得在调用点外面另建一堆临时变量；
  • 没有撤销。想回退，要么在改之前自己存旧值（到处存），要么写一段“反向操作”，而反向操作一旦算错就是新 bug（增益不是加减，模式不是取反）；
  • 调用者知道太多。谁发起的就得知道 device 长什么样、字段叫什么，日后想统一加审计日志，得改每一个调用点。

命令模式的做法很直白：把一次操作连同它的参数、它的接收者、它执行前的旧值，一起封进一个对象。调用者只管 post 和 run；撤销由命令自己负责，因为只有它知道刚才把哪个字段从几改成了几。

02结构：调用者、命令、接收者

三个角色各管一段：调用者管“什么时候执行”，命令管“怎么执行、怎么回退”，接收者只管“被改”。点方框看边界。

命令对象与历史栈command + undo
职责点方框：调用者为什么不必知道 device，旧值为什么必须晚记。

03两份实现

C 版靠“基类结构体放第一位 + ops 函数表”做多态，命令对象就是一段能强转的结构体内存；C++ 版靠虚函数，命令对象由 unique_ptr 在队列与历史之间搬家，所有权始终唯一。

Ccommand.c
/* 命令模式 · C 实现：诊断参数写入队列 + 撤销
 * 编译：gcc -std=c11 -Wall -Wextra command.c -o command
 */
#include <stdio.h>

typedef struct { int gain; int mode; } device_t;

typedef struct cmd cmd_t;
typedef struct {
    void (*exec)(cmd_t *self);
    void (*undo)(cmd_t *self);
} cmd_ops_t;

struct cmd { const cmd_ops_t *ops; };      /* 命令的「类型 + 状态」都在这里 */

/* 具体命令 1：设置增益。exec 时自己记住旧值，undo 才能还原 */
typedef struct {
    cmd_t     base;      /* 必须放第一位：才能把 cmd_t* 安全还原成 set_gain_t* */
    device_t *dev;
    int       value;
    int       old;
} set_gain_t;

static void gain_exec(cmd_t *self)
{
    set_gain_t *c = (set_gain_t *)self;
    c->old = c->dev->gain;
    c->dev->gain = c->value;
    printf("  exec  SetGain(%d)，旧值 %d\n", c->value, c->old);
}

static void gain_undo(cmd_t *self)
{
    set_gain_t *c = (set_gain_t *)self;
    c->dev->gain = c->old;
    printf("  undo  SetGain(%d)，还原成 %d\n", c->value, c->old);
}

static const cmd_ops_t GAIN_OPS = { gain_exec, gain_undo };

/* 具体命令 2：切换模式 */
typedef struct {
    cmd_t     base;
    device_t *dev;
    int       value;
    int       old;
} set_mode_t;

static void mode_exec(cmd_t *self)
{
    set_mode_t *c = (set_mode_t *)self;
    c->old = c->dev->mode;
    c->dev->mode = c->value;
    printf("  exec  SetMode(%d)，旧值 %d\n", c->value, c->old);
}

static void mode_undo(cmd_t *self)
{
    set_mode_t *c = (set_mode_t *)self;
    c->dev->mode = c->old;
    printf("  undo  SetMode(%d)，还原成 %d\n", c->value, c->old);
}

static const cmd_ops_t MODE_OPS = { mode_exec, mode_undo };

/* 调用者：只认 cmd_t*，不知道命令具体做什么 */
typedef struct {
    cmd_t *queued[8];  int qn;
    cmd_t *history[8]; int hn;
} invoker_t;

static void invoker_post(invoker_t *iv, cmd_t *c)
{
    if (iv->qn < 8) iv->queued[iv->qn++] = c;
    printf("  post  队列长度=%d\n", iv->qn);
}

static void invoker_run(invoker_t *iv)
{
    printf("--- 执行队列 ---\n");
    for (int i = 0; i < iv->qn; i++) {
        iv->queued[i]->ops->exec(iv->queued[i]);
        iv->history[iv->hn++] = iv->queued[i];
    }
    iv->qn = 0;
}

static void invoker_undo_last(invoker_t *iv)
{
    if (iv->hn == 0) { printf("  没有可撤销的命令\n"); return; }
    cmd_t *c = iv->history[--iv->hn];
    c->ops->undo(c);
}

int main(void)
{
    device_t dev = { 0, 0 };
    set_gain_t c1 = { { &GAIN_OPS }, &dev, 12, 0 };
    set_gain_t c2 = { { &GAIN_OPS }, &dev, 20, 0 };
    set_mode_t c3 = { { &MODE_OPS }, &dev, 2, 0 };
    invoker_t iv = { .qn = 0, .hn = 0 };

    printf("初始 gain=%d mode=%d\n", dev.gain, dev.mode);
    invoker_post(&iv, (cmd_t *)&c1);
    invoker_post(&iv, (cmd_t *)&c2);
    invoker_post(&iv, (cmd_t *)&c3);
    invoker_run(&iv);
    printf("执行后 gain=%d mode=%d\n", dev.gain, dev.mode);

    printf("--- 撤销两次 ---\n");
    invoker_undo_last(&iv);
    invoker_undo_last(&iv);
    printf("撤销后 gain=%d mode=%d\n", dev.gain, dev.mode);
    printf("历史剩余=%d\n", iv.hn);
    return 0;
}
C++command.cpp
// 命令模式 · C++ 实现：命令对象自带接收者引用，历史栈持有所有权
// 编译：g++ -std=c++17 -Wall -Wextra command.cpp -o command_cpp
#include <cstdio>
#include <deque>
#include <memory>

class Device {
public:
    int gain() const { return gain_; }
    int mode() const { return mode_; }
    void setGain(int v) { gain_ = v; }
    void setMode(int v) { mode_ = v; }
private:
    int gain_ = 0;
    int mode_ = 0;
};

class ICommand {
public:
    virtual ~ICommand() = default;
    virtual void exec() = 0;
    virtual void undo() = 0;
};

class SetGain final : public ICommand {
public:
    SetGain(Device& d, int v) : d_(d), v_(v) {}
    void exec() override { old_ = d_.gain(); d_.setGain(v_); std::printf("  exec  SetGain(%d)，旧值 %d\n", v_, old_); }
    void undo() override { d_.setGain(old_); std::printf("  undo  SetGain(%d)，还原成 %d\n", v_, old_); }
private:
    Device& d_;
    int v_;
    int old_ = 0;
};

class SetMode final : public ICommand {
public:
    SetMode(Device& d, int v) : d_(d), v_(v) {}
    void exec() override { old_ = d_.mode(); d_.setMode(v_); std::printf("  exec  SetMode(%d)，旧值 %d\n", v_, old_); }
    void undo() override { d_.setMode(old_); std::printf("  undo  SetMode(%d)，还原成 %d\n", v_, old_); }
private:
    Device& d_;
    int v_;
    int old_ = 0;
};

// 调用者：只看到 ICommand
class Invoker {
public:
    void post(std::unique_ptr<ICommand> cmd) {
        queue_.push_back(std::move(cmd));
        std::printf("  post  队列长度=%d\n", static_cast<int>(queue_.size()));
    }
    void run() {
        std::printf("--- 执行队列 ---\n");
        for (auto& c : queue_) { c->exec(); history_.push_back(std::move(c)); }
        queue_.clear();
    }
    void undoLast() {
        if (history_.empty()) { std::printf("  没有可撤销的命令\n"); return; }
        history_.back()->undo();
        history_.pop_back();
    }
private:
    std::deque<std::unique_ptr<ICommand>> queue_;
    std::deque<std::unique_ptr<ICommand>> history_;
};

int main() {
    Device dev;
    Invoker iv;
    std::printf("初始 gain=%d mode=%d\n", dev.gain(), dev.mode());

    iv.post(std::make_unique<SetGain>(dev, 12));
    iv.post(std::make_unique<SetGain>(dev, 20));
    iv.post(std::make_unique<SetMode>(dev, 2));
    iv.run();
    std::printf("执行后 gain=%d mode=%d\n", dev.gain(), dev.mode());

    std::printf("--- 撤销两次 ---\n");
    iv.undoLast();
    iv.undoLast();
    std::printf("撤销后 gain=%d mode=%d\n", dev.gain(), dev.mode());
    return 0;
}

04自己排一次队，再撤两次

先按几个命令按钮入队（不会立刻生效），再点“执行队列”。执行过的命令进历史；撤销只从历史尾部取，所以顺序一定是反的。

05C 与 C++：同一套队列，两种所有权

角色C 版C++ 版差别意味着什么
命令接口cmd_ops_t 两个函数指针ICommand 两个纯虚函数都是“一次虚调用”，C 的间接层是显式的函数表
命令对象base 必须放在结构体第一位继承即可，编译器保证布局漏放第一位就是 UB，C 版得靠约定和注释
还原类型(set_gain_t *)selfstatic_cast / 虚分派已到具体类C 的强转没有任何检查，写错就是内存踩踏
创建栈上静态初始化，无分配make_unique，一次堆分配C 版适合固定上限的嵌入式队列
所有权调用者自己保证命令活得够久队列 → 历史，move 转移C++ 版不会出现“撤销时命令已被释放”
撤销都在 exec 里记旧值，undo 里写回两条实现共用同一条规则：旧值晚记、只记一次

06代价与坑

  • 旧值不能在构造时记。命令可能排队几秒才执行；构造时取值，撤销会把设备还原成“入队那一刻”的状态，中间别的写入就白做了。
  • 同一字段连写两次的撤销顺序。先 12 再 20，撤销一次回 12、再撤一次回 0——这依赖历史是严格的 LIFO。想支持“重做”，还得再存一份被撤销的命令。
  • C 版的结构体布局是隐式契约。base 放第一位这条规则，编译器不会检查；一旦有人为了“对齐”把它挪到后面，所有 exec 的强转都失效。用 _Static_assert(offsetof(set_gain_t, base) == 0) 把它变成编译期错误。
  • 撤销不是“反操作”。增益、模式这类赋值型操作，记旧值最稳；如果是“累加”“切换”型操作，反算会引入浮点误差或竞争窗口，同样建议记旧值。
  • 命令里别存裸指针以外的东西。存 device_t* 是必要的；但如果命令还存了“当前总线句柄”“临时缓冲区”，生命周期就会和队列绑在一起，容易泄漏或悬空。
  • 队列满了怎么办。C 版固定 8 格，第 9 条被 if (qn < 8) 悄悄丢掉，但 post 仍打印长度——这是假成功。要么返回失败码，要么覆盖最旧的一条，必须选一个并写进日志。
