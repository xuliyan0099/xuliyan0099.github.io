观察者模式OBSERVER

一次采样，仪表要显示、记录仪要落盘、诊断要判阈值。把这些调用全写在采样函数里，加第四方就得改采样；观察者把“通知谁”变成一个订阅列表。但真正难的不是建表，而是两个边界：通知进行中有人退订，以及订阅者先死了。

两种实现都跑过 C 版对比“朴素/安全” C++ 版 shared_ptr 续命

01问题：通知名单被焊死在采样函数里

最直白的写法是：on_sample() 里依次调 gauge_update()、recorder_write()、diag_check()。能跑，但每次加一个消费者就要改这个函数；三个消费者的顺序、错误码、是否阻塞也全挤在一起。

观察者模式把这段拆成两半：发布者只负责“发生了什么”（notify(ev)），订阅者只负责“我关心时做什么”（onEvent）。发布者不再知道谁在听，订阅者也不需要在编译期被点名。

代价是列表变成了运行期数据，于是出现两个必答题：通知途中有人退订，数组怎么办？订阅者对象已经销毁，回调里那个指针还指哪儿？这两问在本页的演示里都能亲手踩一次。

02结构：发布者只发事件，订阅者各自处理

点方框看职责。虚线表示“订阅关系”，它由订阅者发起，发布者只维护名单。

Subject / Observer 接口 / 三个订阅者structure
职责点方框：发布者、接口、三个订阅者各知道什么、不知道什么。

03两份实现：一份故意先写错

C 版同时给出“朴素”和“安全”两条路径，输出里能直接看到 diag 被跳过；C++ 版用 shared_ptr + 通知前拷贝，顺手解决生命周期。

Cobserver.c
/* 观察者模式 · C 实现：采样总线 + 订阅者表
 * 编译：gcc -std=c11 -Wall -Wextra observer.c -o observer
 * 重点演示「在回调里退订」这一经典崩溃/漏发点。
 */
#include <stdio.h>
#include <string.h>

typedef void (*cb_fn)(void *ctx, const char *event);

typedef struct {
    int   id;            /* -1 表示墓碑：通知中被退订，等通知结束再压缩 */
    char  name[20];
    cb_fn fn;
    void *ctx;
} sub_t;

typedef struct {
    sub_t slot[8];
    int   n;
    int   depth;         /* 正在通知的层数 */
} bus_t;

typedef struct { bus_t *bus; int id; } peer_t;

static void bus_attach(bus_t *b, int id, const char *name, cb_fn fn, void *ctx)
{
    if (b->n >= 8) return;
    b->slot[b->n].id = id;
    snprintf(b->slot[b->n].name, sizeof b->slot[b->n].name, "%s", name);
    b->slot[b->n].fn = fn;
    b->slot[b->n].ctx = ctx;
    b->n++;
}

static void bus_compact(bus_t *b)
{
    int w = 0;
    for (int i = 0; i < b->n; i++)
        if (b->slot[i].id >= 0) b->slot[w++] = b->slot[i];
    b->n = w;
}

/* 朴素实现：立刻前移数组 */
static void bus_detach_raw(bus_t *b, int id)
{
    for (int i = 0; i < b->n; i++)
        if (b->slot[i].id == id) {
            for (int k = i; k < b->n - 1; k++) b->slot[k] = b->slot[k + 1];
            b->n--;                     /* 没有 i--，外层循环下一次就跳过了一个 */
            return;
        }
}

/* 朴素通知：边遍历边让 b->n 变化 */
static void bus_notify_raw(bus_t *b, const char *ev)
{
    for (int i = 0; i < b->n; i++)
        b->slot[i].fn(b->slot[i].ctx, ev);
}

/* 安全实现：通知中只打墓碑，不移动元素 */
static void bus_detach_safe(bus_t *b, int id)
{
    int i;
    for (i = 0; i < b->n; i++) if (b->slot[i].id == id) break;
    if (i == b->n) return;
    if (b->depth > 0) { b->slot[i].id = -1; return; }
    for (int k = i; k < b->n - 1; k++) b->slot[k] = b->slot[k + 1];
    b->n--;
}

/* 安全通知：先拍 id 快照，再按 id 找还活着的槽 */
static void bus_notify_safe(bus_t *b, const char *ev)
{
    int ids[8];
    int m = b->n;
    for (int i = 0; i < m; i++) ids[i] = b->slot[i].id;

    b->depth++;
    for (int i = 0; i < m; i++) {
        for (int k = 0; k < b->n; k++)
            if (b->slot[k].id == ids[i]) {
                b->slot[k].fn(b->slot[k].ctx, ev);
                break;
            }
    }
    b->depth--;
    if (b->depth == 0) bus_compact(b);
}

/* 三个订阅者 */
static void cb_gauge(void *ctx, const char *ev)
{
    (void)ctx;
    printf("  gauge    <- %s\n", ev);
}

static void cb_diag(void *ctx, const char *ev)
{
    (void)ctx;
    printf("  diag     <- %s\n", ev);
}

static void cb_recorder_raw(void *ctx, const char *ev)
{
    peer_t *p = ctx;
    printf("  recorder <- %s（收到后退订自己）\n", ev);
    bus_detach_raw(p->bus, p->id);
}

static void cb_recorder_safe(void *ctx, const char *ev)
{
    peer_t *p = ctx;
    printf("  recorder <- %s（收到后退订自己）\n", ev);
    bus_detach_safe(p->bus, p->id);
}

int main(void)
{
    bus_t raw = { .n = 0, .depth = 0 };
    bus_t safe = { .n = 0, .depth = 0 };
    peer_t r1 = { &raw, 2 }, r2 = { &safe, 2 };

    printf("朴素版本：\n");
    bus_attach(&raw, 1, "gauge", cb_gauge, NULL);
    bus_attach(&raw, 2, "recorder", cb_recorder_raw, &r1);
    bus_attach(&raw, 3, "diag", cb_diag, NULL);
    bus_notify_raw(&raw, "speed=62");
    printf("  剩余订阅者=%d（diag 被跳过）\n\n", raw.n);

    printf("安全版本：\n");
    bus_attach(&safe, 1, "gauge", cb_gauge, NULL);
    bus_attach(&safe, 2, "recorder", cb_recorder_safe, &r2);
    bus_attach(&safe, 3, "diag", cb_diag, NULL);
    bus_notify_safe(&safe, "speed=62");
    bus_notify_safe(&safe, "speed=63");
    printf("  剩余订阅者=%d\n", safe.n);
    return 0;
}
C++observer.cpp
// 观察者模式 · C++ 实现：shared_ptr 订户 + 通知时拷贝列表
// 编译：g++ -std=c++17 -Wall -Wextra observer.cpp -o observer_cpp
#include <cstdio>
#include <memory>
#include <string>
#include <vector>

class Observer {
public:
    virtual ~Observer() = default;
    virtual void onEvent(const std::string& ev) = 0;
};

class Subject {
public:
    int attach(std::shared_ptr<Observer> o) {
        slots_.push_back(Slot{ next_id_++, std::move(o) });
        return slots_.back().id;
    }

    // 通知中退订：只清空指针，不移动元素
    void detach(int id) {
        for (auto& s : slots_) {
            if (s.id != id) continue;
            s.obs.reset();
            if (depth_ == 0) compact();
            return;
        }
    }

    void notify(const std::string& ev) {
        auto snapshot = slots_;          // 关键：拷贝，通知期间列表不被改动
        ++depth_;
        for (auto& s : snapshot)
            if (s.obs) s.obs->onEvent(ev);
        --depth_;
        if (depth_ == 0) compact();
    }

private:
    struct Slot { int id; std::shared_ptr<Observer> obs; };
    std::vector<Slot> slots_;
    int next_id_ = 1;
    int depth_ = 0;

    void compact() {
        std::vector<Slot> live;
        for (auto& s : slots_) if (s.obs) live.push_back(std::move(s));
        slots_ = std::move(live);
    }
};

class Gauge final : public Observer {
public:
    void onEvent(const std::string& ev) override { std::printf("  gauge    <- %s\n", ev.c_str()); }
};

class Diag final : public Observer {
public:
    void onEvent(const std::string& ev) override { std::printf("  diag     <- %s\n", ev.c_str()); }
};

class Recorder final : public Observer {
public:
    explicit Recorder(Subject& s) : s_(s) {}
    void bind(int id) { id_ = id; }
    void onEvent(const std::string& ev) override {
        std::printf("  recorder <- %s（收到后退订自己）\n", ev.c_str());
        if (id_ > 0) s_.detach(id_);
    }
private:
    Subject& s_;
    int id_ = 0;
};

int main() {
    Subject bus;
    auto gauge = std::make_shared<Gauge>();
    auto rec   = std::make_shared<Recorder>(bus);
    auto diag  = std::make_shared<Diag>();

    bus.attach(gauge);
    rec->bind(bus.attach(rec));
    bus.attach(diag);

    std::printf("安全版本：\n");
    bus.notify("speed=62");
    bus.notify("speed=63");
    return 0;
}

05同一组角色，两种写法

环节C 版C++ 版要防的事
存订户sub_t slot[8] 固定数组vector<Slot> 动态容量上限 / 扩容时机
订阅attach(id, cb, ctx)attach(shared_ptr) 返回 id重复订阅、id 复用
通知中退订打墓碑 id = -1，通知结束再压缩obs.reset()，通知结束再压缩边遍历边移动元素
通知顺序按数组顺序，先拍 id 快照拷贝 slots_ 再遍历不许依赖顺序做业务
生命周期无保护：ctx 悬垂就是未定义行为shared_ptr 在通知期间续命订户先于发布者销毁
为什么 C++ 版能顺手解决生命周期？拷贝 slots_ 时，shared_ptr 的引用计数 +1；即使订户在回调里退订，对象也活到本轮通知结束。C 版要等价，得自己做引用计数，或者约定“销毁前必须先 detach”。

06代价与坑

1. 通知是同步的。notify() 返回时所有回调已经跑完。订阅者里做阻塞 IO、等锁、甚至再触发一次 notify，都会把发布者卡住或造成重入。实时系统里通常改成“入队 + 另一线程分发”。
2. 退订要“延迟生效”。本轮通知已经决定要叫谁，中途删掉会让后面的人收不到（本页演示的正是这个）。墓碑 + 结束压缩是最小改动的解法。
3. 订阅者的生命周期比接口更难。C 里没有 shared_ptr，要么约定销毁顺序，要么在发布者里存“是否还活着”的仲裁位，要么用句柄（不返回裸指针）。
4. 别用通知顺序表达业务。“先仪表后诊断”这种隐含依赖，在加一个订阅者时就崩了。需要顺序就显式分两级总线，或在事件里带序号。
5. 事件对象别传栈上临时值的地址。C 版 ev 是 const char*，回调里若存起来供后续使用，必须自己拷贝；C++ 版用 std::string 按值传，省掉这一类悬垂。
