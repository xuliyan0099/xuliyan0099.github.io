工厂模式FACTORY

配置文件里写着 sensor = "sht3x"，代码里却没有一个 if (name == "sht3x")。答案是一张注册表：每个驱动在自己被链接进来的那一刻就把“名字 → 构造函数”登记进去，上层只按字符串查表，查不到就返回空，由它决定怎么降级。

名字 → 构造函数 自注册在 main 之前 未知名字返回 NULL/nullptr

01问题：新增一个型号就要改 switch

最直接的写法是一个大 if/else：名字是 "sht3x" 就 new Sht3x，是 "ntc" 就 new Ntc。它的问题不在“难写”，而在它必须知道所有产品：

  • 每加一个驱动，这个 switch 就要改一次，哪怕新驱动跟这段代码毫无关系；
  • 产品构造逻辑（哪个地址、哪个引脚、要不要先 init）也被写进了选择逻辑里，越写越厚；
  • 如果驱动是可选编译的（不同车型带不同传感器），switch 里还得跟一堆 #ifdef。

注册表把“认识所有产品”这件事反转了：不是工厂认识产品，而是产品在启动时告诉工厂“我叫什么、怎么创建我”。工厂退化成一张表和一次字符串查找，新增驱动只需新增一个文件，连 main 都不用动。

02结构：一张表、两个方向

向下：工厂按名字查表，调构造函数拿到产品；向上：产品在静态初始化阶段把自己登记进表。点方框看谁在什么时候做了什么。

注册表式工厂registry + self-registration
职责点方框：为什么工厂可以完全不认识产品，为什么未知名字不报错而是返回空。

03两份实现

C 版是一张固定长度的结构体数组 + driver_t *(*)(void) 函数指针；C++ 版是 map<string, std::function<unique_ptr<Driver>()>>，自注册收在一个模板里。两边的“登记先于创建”完全一样。

Cfactory.c
/* 工厂模式 · C 实现：按名字创建驱动（注册表 + 构造函数指针）
 * 编译：gcc -std=c11 -Wall -Wextra factory.c -o factory
 */
#include <stdio.h>
#include <string.h>

typedef struct driver driver_t;
typedef struct {
    int         (*init)(driver_t *self);
    const char *(*name)(driver_t *self);
} driver_ops_t;

struct driver { const driver_ops_t *ops; };

typedef driver_t *(*create_fn)(void);

#define MAX_REG 8
typedef struct { char name[16]; create_fn create; } entry_t;

static entry_t g_reg[MAX_REG];
static int     g_reg_n = 0;

static int driver_register(const char *name, create_fn fn)
{
    if (g_reg_n >= MAX_REG) return -1;
    snprintf(g_reg[g_reg_n].name, sizeof g_reg[g_reg_n].name, "%s", name);
    g_reg[g_reg_n].create = fn;
    g_reg_n++;
    printf("  register %-6s  (注册表长度=%d)\n", name, g_reg_n);
    return 0;
}

static driver_t *driver_create(const char *name)
{
    for (int i = 0; i < g_reg_n; i++)
        if (strcmp(g_reg[i].name, name) == 0)
            return g_reg[i].create();
    return NULL;
}

/* 具体产品 1 */
typedef struct { driver_t base; int addr; } sht_drv_t;

static int sht_init(driver_t *self) {
    sht_drv_t *d = (sht_drv_t *)self;
    printf("    sht3x init @0x%02X\n", d->addr);
    return 0;
}
static const char *sht_name(driver_t *self) { (void)self; return "sht3x"; }
static const driver_ops_t SHT_OPS = { sht_init, sht_name };
static driver_t *sht_new(void) { static sht_drv_t d = { { &SHT_OPS }, 0x44 }; return &d.base; }

/* 具体产品 2 */
typedef struct { driver_t base; int pin; } ntc_drv_t;

static int ntc_init(driver_t *self) {
    ntc_drv_t *d = (ntc_drv_t *)self;
    printf("    ntc init on pin %d\n", d->pin);
    return 0;
}
static const char *ntc_name(driver_t *self) { (void)self; return "ntc"; }
static const driver_ops_t NTC_OPS = { ntc_init, ntc_name };
static driver_t *ntc_new(void) { static ntc_drv_t d = { { &NTC_OPS }, 12 }; return &d.base; }

/* 链接期自注册：启动时自动登记，不用改 main */
static void auto_reg_ntc(void) __attribute__((constructor));
static void auto_reg_ntc(void) { driver_register("ntc", ntc_new); }

int main(void)
{
    printf("启动后注册表：\n");
    for (int i = 0; i < g_reg_n; i++) printf("  [%d] %s\n", i, g_reg[i].name);
    printf("\n");

    driver_register("sht3x", sht_new);

    const char *wanted[] = { "sht3x", "ntc", "bme280" };
    for (unsigned i = 0; i < 3; i++) {
        printf("create(\"%s\"):\n", wanted[i]);
        driver_t *d = driver_create(wanted[i]);
        if (!d) { printf("    未注册，返回 NULL（由上层决定降级）\n"); continue; }
        d->ops->init(d);
        printf("    name=%s\n", d->ops->name(d));
    }
    return 0;
}
C++factory.cpp
// 工厂模式 · C++ 实现：注册表 + std::function + 自注册模板
// 编译：g++ -std=c++17 -Wall -Wextra factory.cpp -o factory_cpp
#include <cstdio>
#include <functional>
#include <map>
#include <memory>
#include <string>
#include <vector>

class Driver {
public:
    virtual ~Driver() = default;
    virtual const char* name() const = 0;
    virtual bool init() = 0;
};

class Sht3x final : public Driver {
public:
    const char* name() const override { return "sht3x"; }
    bool init() override { std::printf("    sht3x init @0x44\n"); return true; }
};

class Ntc final : public Driver {
public:
    const char* name() const override { return "ntc"; }
    bool init() override { std::printf("    ntc init on pin 12\n"); return true; }
};

class DriverFactory {
public:
    using Creator = std::function<std::unique_ptr<Driver>()>;

    static DriverFactory& instance() {      // 函数内静态：规避初始化顺序问题
        static DriverFactory f;
        return f;
    }

    void reg(const std::string& name, Creator c) {
        map_[name] = std::move(c);
        std::printf("  register %-6s  (注册表长度=%d)\n", name.c_str(), static_cast<int>(map_.size()));
    }

    std::unique_ptr<Driver> create(const std::string& name) const {
        auto it = map_.find(name);
        if (it == map_.end()) return nullptr;
        return it->second();
    }

    std::vector<std::string> names() const {
        std::vector<std::string> v;
        for (const auto& kv : map_) v.push_back(kv.first);
        return v;
    }

private:
    std::map<std::string, Creator> map_;
};

// 自注册：定义一个静态对象，构造即登记
template <class T>
struct AutoReg {
    explicit AutoReg(const char* name) {
        DriverFactory::instance().reg(name, [] { return std::make_unique<T>(); });
    }
};

static AutoReg<Ntc>   g_reg_ntc("ntc");
static AutoReg<Sht3x> g_reg_sht("sht3x");

int main() {
    std::printf("启动后注册表：\n");
    for (const auto& n : DriverFactory::instance().names()) std::printf("  %s\n", n.c_str());
    std::printf("\n");

    const char* wanted[] = { "sht3x", "ntc", "bme280" };
    for (const char* w : wanted) {
        std::printf("create(\"%s\"):\n", w);
        auto d = DriverFactory::instance().create(w);
        if (!d) { std::printf("    未注册，返回 nullptr（由上层决定降级）\n"); continue; }
        d->init();
        std::printf("    name=%s\n", d->name());
    }
    return 0;
}

05C 与 C++：一张表，两种造法

角色C 版C++ 版差别意味着什么
产品接口driver_ops_t（init / name）Driver 纯虚类同一层间接调用，C 的 ops 表可以 const 放 flash
注册项entry_t{ char name[16]; create_fn }map<string, Creator>C 定长无分配，名字超过 15 字符会被截断
自注册__attribute__((constructor))static AutoReg<T>前者是 GCC 扩展，后者是标准 C++，但都依赖“静态初始化先于 main”
create 返回内部静态对象指针（每次都同一个）unique_ptr，每次都新建这是最实质的差别：C 版其实是“按名字取单例”
未注册NULLnullptr都不报错、不 abort，把降级决策留给上层
扩容MAX_REG 8，满了返回 -1map 自动扩容C 版适合启动期固定集合，运行时动态加要另做内存池

06代价与坑

  • 静态初始化顺序。C++ 里两个 AutoReg 对象和工厂单例的构造顺序跨编译单元是不确定的。把工厂写成 instance() 里的函数内静态，能保证“第一次用到时才构造”，避免在工厂还没构造好时就去注册。
  • 链接器会把自注册对象裁掉。--gc-sections / -ffunction-sections 下，没有被任何代码引用的静态对象可能整个被丢弃，表现是“注册表里莫名其妙少了一个驱动”。要么加 KEEP 段，要么改成显式注册。
  • C 版 create 不是“新建”。sht_new 返回的是函数内静态对象，两次 create 得到同一块内存。若上层以为各自独立、改了其中一个的配置，另一个也变了。要真新建就得给分配器或对象池。
  • 名字匹配是严格的。strcmp 区分大小写，配置里写成 "SHT3X" 或带空格就查不到。要么统一小写化，要么在注册处做别名。
  • 注册表是全局可变状态。运行时并发注册要加锁；最稳的策略是“只在启动阶段注册，之后只读”。
  • 未知名字不是错误。返回 NULL 是刻意的：不同车型可以没有某个传感器。但上层必须处理，不能直接解引用。
  • 销毁顺序。C++ 版 unique_ptr 走虚析构没问题；如果产品里持有互斥量、DMA 通道，要保证工厂/注册表的析构不早于产品使用。
