    建造者模式BUILDER

    拼一帧 CAN 报文有七八个字段，顺序错了不一定报错，可能只是 CRC 不对、DLC 不对。建造者把“拼装”变成一条有步骤的流水线：每一步只改一个字段，最后 build() 才交出成品；中途可以校验、可以复用，也可以随时丢弃重来。

    步骤可查成品只交付一次复用 builder 不残留上一帧

    01问题：一帧报文七八个字段，顺序全靠记

    常见写法是一个大初始化：

    frame_t f = { id, dlc, {b0,b1,b2}, crc };

    字段一多就出三类问题：

    • 顺序靠记。CRC 必须在数据之后算；有人先算 CRC 再填数据，台架上偶尔能过，因为数据恰好没变。
    • 复用会串味。同一个栈上的 frame_t 反复用，上一帧的 data[3..7] 没清，DLC 又写成 8，发出去就是脏数据。
    • 中途失败没法收场。拼到一半发现 id 非法，只能 return，已写进去的半成品留在对象里。

    建造者把每个字段的写入变成一个显式步骤，并提供 start() 清底、build() 交付。调用方看到的是“开始 → 填 id → 填数据 → 算 CRC → 交付”，而不是一堆裸赋值。

    02结构：步骤可查、成品只交付一次

    四个角色排成一条线：调用方发起，builder 累积，成品交付给发送层。

    一帧报文的建造流水线builder
    职责点方框：builder 负责累积，frame 只负责被交付。

    03两份实现

    C 版是一组 b_* 函数 + 结构体；C++ 版把每个步骤做成返回 Builder& 的链式方法。

    Cbuilder.c
    /* 建造者：把“拼一帧”拆成可检查的步骤，最后一次性交付 */
#include <stdio.h>
#include <stddef.h>

typedef struct {
    unsigned id;
    int dlc;
    unsigned char data[8];
    unsigned crc;
} frame_t;

typedef struct { frame_t f; } builder_t;

static void b_start(builder_t *b)          { b->f.id = 0; b->f.dlc = 0; b->f.crc = 0; }
static void b_id(builder_t *b, unsigned v) { b->f.id = v; }
static void b_data(builder_t *b, const unsigned char *d, int n)
{
    if (n > 8) n = 8;
    b->f.dlc = n;
    for (int i = 0; i < n; i++) b->f.data[i] = d[i];
}
static void b_crc(builder_t *b)
{
    unsigned sum = b->f.id;
    for (int i = 0; i < b->f.dlc; i++) sum += b->f.data[i];
    b->f.crc = sum & 0xFFu;
}
static frame_t b_build(const builder_t *b) { return b->f; }

static void dump(const frame_t *f)
{
    printf("id=0x%X dlc=%d data=[", f->id, f->dlc);
    for (int i = 0; i < f->dlc; i++) printf("%s%02X", i ? " " : "", f->data[i]);
    printf("] crc=0x%02X\n", f->crc);
}

int main(void)
{
    builder_t b;
    unsigned char payload[8] = { 0x10, 0x20, 0x30, 0x40, 0x50, 0x60, 0x70, 0x80 };

    b_start(&b);
    b_id(&b, 0x18FF50E5u);
    b_data(&b, payload, 3);
    b_crc(&b);
    frame_t f1 = b_build(&b);
    printf("frame1 ");
    dump(&f1);

    b_start(&b);                 /* 复用同一个 builder，不残留上一帧 */
    b_id(&b, 0x7DFu);
    b_data(&b, payload, 8);      /* 超长被截到 8 字节 */
    b_crc(&b);
    frame_t f2 = b_build(&b);
    printf("frame2 ");
    dump(&f2);
    printf("两次建造独立：frame1 dlc=%d, frame2 dlc=%d\n", f1.dlc, f2.dlc);
    return 0;
}

    C++builder.cpp
    // 建造者 · C++：链式调用，build() 只交付一次
// 编译：g++ -std=c++17 -Wall -Wextra builder.cpp -o builder_cpp
#include <cstdio>
#include <array>

struct Frame {
    unsigned id = 0;
    int dlc = 0;
    std::array<unsigned char, 8> data{};
    unsigned crc = 0;
};

class Builder {
public:
    Builder &start() { f_ = Frame{}; return *this; }
    Builder &id(unsigned v) { f_.id = v; return *this; }
    Builder &data(const unsigned char *d, int n)
    {
        if (n > 8) n = 8;
        f_.dlc = n;
        for (int i = 0; i < n; i++) f_.data[static_cast<size_t>(i)] = d[i];
        return *this;
    }
    Builder &crc()
    {
        unsigned sum = f_.id;
        for (int i = 0; i < f_.dlc; i++) sum += f_.data[static_cast<size_t>(i)];
        f_.crc = sum & 0xFFu;
        return *this;
    }
    Frame build() const { return f_; }
private:
    Frame f_;
};

static void dump(const Frame &f)
{
    std::printf("id=0x%X dlc=%d data=[", f.id, f.dlc);
    for (int i = 0; i < f.dlc; i++) std::printf("%s%02X", i ? " " : "", f.data[static_cast<size_t>(i)]);
    std::printf("] crc=0x%02X\n", f.crc);
}

int main()
{
    unsigned char payload[8] = { 0x10, 0x20, 0x30, 0x40, 0x50, 0x60, 0x70, 0x80 };
    Builder b;
    Frame f1 = b.start().id(0x18FF50E5u).data(payload, 3).crc().build();
    std::printf("frame1 ");
    dump(f1);

    Frame f2 = b.start().id(0x7DFu).data(payload, 8).crc().build();
    std::printf("frame2 ");
    dump(f2);
    std::printf("两次建造独立：frame1 dlc=%d, frame2 dlc=%d\n", f1.dlc, f2.dlc);
    return 0;
}

    05C 与 C++：两种链式

    角色C 版C++ 版差别意味着什么
    半成品放哪builder_t.f 内嵌Builder::f_ 私有成员都在 builder 里，调用方拿不到半成品
    步骤接口b_id() / b_data() / b_crc()链式 id().data().crc()C 靠命名约束顺序，C++ 靠返回值把顺序写进表达式
    重置b_start() 显式清start() 重新赋值 Frame{}C 要记得清每个字段；C++ 赋值一次全清
    交付按值返回 frame_t按值返回 Frame都是副本，交付后改 builder 不影响成品
    超长数据b_data() 里截到 8 字节同样截到 8 字节规则一致，但都要在文档里写明，别让调用方猜

    06代价与坑

    • CRC 必须在数据之后。链式调用看起来自由，但 crc() 依赖 dlc 与数据；把它写在最前，算出来一定是错的。
    • build() 之后别再改成品。按值返回意味着成品是副本；如果后续要改，重新 start 再建一帧，不要绕过 builder 直接改字段。
    • 数据截断要报错。超过 8 字节静默截断，调用方以为全发出去了；至少要返回状态或日志，不能只截不说。
    • builder 不是线程安全的。一个 builder 只给一个任务用；多线程共用要加锁，或者每个线程各建一个。
    • 步骤别无限膨胀。字段超过十几个、还带条件分支时，考虑把“帧描述”做成表驱动，而不是继续加方法。
