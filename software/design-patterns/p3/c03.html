    组合模式COMPOSITE

    资源管理器里，文件夹可以装文件，也可以装文件夹。如果“遍历”只认文件夹、“统计”只认文件，调用方就得一路判断类型。组合模式让枝和叶实现同一个接口：对整棵树调用一次，内部自己递归。

    枝与叶同接口递归收敛在一处C 用 is_dir 标记 · C++ 用虚函数

    01问题：坏代码长什么样

    坏写法是调用方自己分类型递归：

    if (node->is_dir)
    for (each kid) total(kid);
else
    sum += node->size;

    这段逻辑会在“统计大小”“查找文件”“生成树形图”里各抄一遍，而且每处都要记得处理空目录。

    组合把 total() 放到节点自己身上：目录的 total 累加孩子，文件的 total 返回自身大小。调用方只写 total(root)。

    02结构：枝递归、叶终止

    左边是调用方，中间是目录节点，右边是文件叶子。

    目录递归孩子，文件返回自身structure
    职责枝负责“继续往下”，叶负责“到此为止”。

    03两份实现：C 与 C++

    C 版用 is_dir 标记区分枝与叶，孩子数组固定容量；C++ 版用虚函数和多态容器。

    Ccomposite.c
    /* 组合 · 文件与文件夹
 * gcc -std=c11 -Wall -Wextra composite.c -o composite
 */
#include <stdio.h>
#include <string.h>

typedef struct node node_t;
struct node {
    char name[16];
    int  is_dir;
    int  size;                 /* 文件大小；目录忽略 */
    node_t *kids[4];
    int  nkids;
};

static node_t *add(node_t *dir, node_t *kid)
{
    if (dir->nkids < 4) dir->kids[dir->nkids++] = kid;
    return dir;
}

static int total(const node_t *n)
{
    if (!n->is_dir) return n->size;
    int sum = 0;
    for (int i = 0; i < n->nkids; i++) sum += total(n->kids[i]);
    return sum;
}

static void list(const node_t *n, int depth)
{
    for (int i = 0; i < depth; i++) printf("  ");
    printf("%s%s\n", n->is_dir ? "[D] " : "    ", n->name);
    if (n->is_dir)
        for (int i = 0; i < n->nkids; i++) list(n->kids[i], depth + 1);
}

int main(void)
{
    node_t root = { "root", 1, 0, { 0 }, 0 };
    node_t src  = { "src",  1, 0, { 0 }, 0 };
    node_t a    = { "a.c",  0, 120, { 0 }, 0 };
    node_t b    = { "b.c",  0, 80,  { 0 }, 0 };
    node_t read = { "README", 0, 10, { 0 }, 0 };

    add(&src, &a); add(&src, &b);
    add(&root, &src); add(&root, &read);

    list(&root, 0);
    printf("总大小 = %d 字节\n", total(&root));
    return 0;
}
    RUN./composite.exe
    [D] root
  [D] src
        a.c
        b.c
      README
总大小 = 210 字节
    C++composite.cpp
    // 组合 · 文件与文件夹
// g++ -std=c++17 -Wall -Wextra composite.cpp -o composite_cpp
#include <cstdio>
#include <memory>
#include <string>
#include <vector>

class Node {
public:
    Node(std::string name, bool dir, int size = 0)
        : name_(std::move(name)), dir_(dir), size_(size) {}
    virtual ~Node() = default;

    void add(std::unique_ptr<Node> kid) { kids_.push_back(std::move(kid)); }
    int total() const {
        if (!dir_) return size_;
        int sum = 0;
        for (const auto& k : kids_) sum += k->total();
        return sum;
    }
    void list(int depth = 0) const {
        for (int i = 0; i < depth; i++) std::printf("  ");
        std::printf("%s%s\n", dir_ ? "[D] " : "    ", name_.c_str());
        for (const auto& k : kids_) k->list(depth + 1);
    }
private:
    std::string name_;
    bool dir_;
    int size_;
    std::vector<std::unique_ptr<Node>> kids_;
};

int main()
{
    auto root = std::make_unique<Node>("root", true);
    auto src = std::make_unique<Node>("src", true);
    src->add(std::make_unique<Node>("a.c", false, 120));
    src->add(std::make_unique<Node>("b.c", false, 80));
    root->add(std::move(src));
    root->add(std::make_unique<Node>("README", false, 10));

    root->list();
    std::printf("总大小 = %d 字节\n", root->total());
    return 0;
}
    RUN./composite_cpp.exe
    [D] root
  [D] src
        a.c
        b.c
      README
总大小 = 210 字节

    05C 与 C++ 对照

    角色C 版C++ 版差别意味着什么
    枝与叶怎么分is_dir 字段不同子类 + 虚函数C 靠标记，C++ 靠类型
    递归在哪节点的 total() 里节点的 total() 里都不在调用方
    孩子容量固定数组 4 个vector<unique_ptr>C 版示例简化，真实实现要动态
    空目录返回 0返回 0必须显式处理，别返回未初始化值

    06代价与坑

    参考教程：refactoringguru.cn

    • 枝和叶接口不一致一旦调用方要 if (is_dir)，组合就失效了。
    • 递归没有终止条件叶节点必须明确“没有孩子”，否则空指针递归。
    • 父节点持有孩子又共享所有权C++ 用 unique_ptr 明确父子；需要共享时再谈引用计数。
