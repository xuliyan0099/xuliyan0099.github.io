    适配器模式ADAPTER

    旧接口只接受“半径”，新零件偏偏是“边长”。改旧接口会影响所有调用方，改新零件又违反它自己的规格。适配器夹在中间：对外仍是旧接口，对内把方钉翻译成等效半径。

    旧接口不动新零件不改C 用包装结构 · C++ 用类继承 + 组合

    01问题：坏代码长什么样

    坏写法是让调用方自己换算：

    if (peg_kind == SQUARE)
    ok = round_fits(&hole, peg.width * 2 / 3);
else
    ok = round_fits(&hole, peg.radius);

    换算规则散落到每个调用点；再来一种六角钉，所有地方都要补分支。

    适配器把换算收在一处：它实现旧接口（对外暴露 radius()），内部持有方钉，把边长翻译成等效半径。调用方仍然只认圆孔。

    02结构：旧接口、适配器、新零件

    左边是圆孔，中间是适配器，右边是方钉。

    旧接口在前，翻译在中间，新零件在后structure
    职责适配器是唯一知道两种规格怎么换算的对象。

    03两份实现：C 与 C++

    C 版适配器是一个包装结构体，靠函数读内部方钉；C++ 版用继承拿到旧接口、用组合持有方钉。

    Cadapter.c
    /* 适配器 · 方钉插圆孔
 * gcc -std=c11 -Wall -Wextra adapter.c -o adapter
 */
#include <stdio.h>

typedef struct { int radius; } round_hole_t;
typedef struct { int width;  } square_peg_t;

/* 旧接口：只认圆钉 */
static int round_fits(const round_hole_t *h, int peg_radius)
{
    return peg_radius <= h->radius;
}

/* 适配器：把方钉包一层，暴露“半径” */
typedef struct { square_peg_t peg; } square_adapter_t;

static int adapter_radius(const square_adapter_t *a)
{
    /* 方钉对角线一半，才能塞进圆孔 */
    return (a->peg.width * 2) / 3;
}

int main(void)
{
    round_hole_t hole = { 5 };
    square_peg_t small = { 6 };
    square_peg_t big   = { 10 };

    printf("圆孔半径 %d\n", hole.radius);

    square_adapter_t a1 = { small };
    square_adapter_t a2 = { big };
    printf("方钉 6  -> 适配半径 %d -> %s\n", adapter_radius(&a1),
           round_fits(&hole, adapter_radius(&a1)) ? "塞得进" : "塞不进");
    printf("方钉 10 -> 适配半径 %d -> %s\n", adapter_radius(&a2),
           round_fits(&hole, adapter_radius(&a2)) ? "塞得进" : "塞不进");
    return 0;
}
    RUN./adapter.exe
    圆孔半径 5
方钉 6  -> 适配半径 4 -> 塞得进
方钉 10 -> 适配半径 6 -> 塞不进
    C++adapter.cpp
    // 适配器 · 方钉插圆孔
// g++ -std=c++17 -Wall -Wextra adapter.cpp -o adapter_cpp
#include <cstdio>

class RoundHole {
public:
    explicit RoundHole(int r) : r_(r) {}
    int radius() const { return r_; }
    bool fits(int pegRadius) const { return pegRadius <= r_; }
private:
    int r_;
};

class SquarePeg {
public:
    explicit SquarePeg(int w) : w_(w) {}
    int width() const { return w_; }
private:
    int w_;
};

class SquarePegAdapter {
public:
    explicit SquarePegAdapter(const SquarePeg& p) : peg_(p) {}
    int radius() const { return (peg_.width() * 2) / 3; }
private:
    SquarePeg peg_;
};

int main()
{
    RoundHole hole(5);
    SquarePeg small(6), big(10);
    SquarePegAdapter a1(small), a2(big);

    std::printf("圆孔半径 %d\n", hole.radius());
    std::printf("方钉 6  -> 适配半径 %d -> %s\n", a1.radius(),
                hole.fits(a1.radius()) ? "塞得进" : "塞不进");
    std::printf("方钉 10 -> 适配半径 %d -> %s\n", a2.radius(),
                hole.fits(a2.radius()) ? "塞得进" : "塞不进");
    return 0;
}
    RUN./adapter_cpp.exe
    圆孔半径 5
方钉 6  -> 适配半径 4 -> 塞得进
方钉 10 -> 适配半径 6 -> 塞不进

    两根方钉，一次塞得进、一次塞不进圆孔半径 5，方钉边长 6 和 10。

    05C 与 C++ 对照

    角色C 版C++ 版差别意味着什么
    适配器形态包装结构 + 换算函数继承旧接口 + 组合新零件C 没有继承，用结构嵌套
    调用方看到adapter_radius()radius() 虚函数对外都是旧接口
    换算规则放哪适配器函数里适配器成员函数里只此一处，改规则不改调用方
    额外成本一次函数调用 + 一次换算一次虚调用 + 一次换算都很薄，别在适配器里做重活

    06代价与坑

    参考教程：refactoringguru.cn

    • 适配器变成业务层它只做接口翻译，不要在适配器里加缓存、重试、权限。
    • 换算有损还假装无损方钉对角线近似成半径会失真；需要精确时应在接口层说明误差。
    • 双向适配两个方向都要就写两个适配器，不要一个类同时继承两边，容易循环依赖。
