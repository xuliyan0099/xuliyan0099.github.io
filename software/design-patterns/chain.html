职责链模式CHAIN

把“检查”拆成独立的一节一节，串成一条链：每一节只判断自己关心的那一件事，通过就把报文交给下一节，不通过就就地终止。报文从哪一节被丢，看日志就知道；想加一节过滤，只改装配，不动任何已有节点。

四个节点 · 四种丢弃点 通过 = 转 next 终止 = 不碰 next

01问题：嵌套 if 会越写越深

报文过滤最直觉的写法是层层嵌套：长度对了才查 CRC，CRC 对了才查白名单，白名单过了才交给业务。三个检查就已经三层缩进，再加“频率限制”“报文年龄”“安全访问”就往右顶到屏幕外面。

更深一层的问题是顺序变成了代码结构：想调换 CRC 与白名单的先后，得把整块代码搬来搬去；想按车型装配不同的过滤组合，只能复制函数再改；想在丢弃时统一记一笔，得在每个 return 前都写一遍。

职责链把顺序变成数据：每个检查是一个节点，节点只知道“我通过就转给 next”。装配处把 len → crc → white → app 串起来，顺序一目了然，换顺序就换链。

02结构：一条单向链，四道关

每节节点做两件事：判断自己，决定是否调用 next。点方框看每一节“知道什么、不知道什么”。

CAN 报文过滤链len → crc → white → app
职责点方框：每节只判断一件事，终止与放行由同一处逻辑决定。

03两份实现

C 版把链做成结构体里的 next 指针，转发是节点自己写的一句递归调用；C++ 版把“转发”提到基类 run() 里，子类只实现 check()，从结构上保证每个节点都不会忘记转 next。

Cchain.c
/* 责任链模式 · C 实现：报文过滤管道
 * 编译：gcc -std=c11 -Wall -Wextra chain.c -o chain
 * 每个处理器返回 0 = 放行给下一个，1 = 就地终止。
 */
#include <stdio.h>

typedef struct { int id; int len; unsigned char crc; } frame_t;

typedef struct handler handler_t;
typedef struct {
    int (*handle)(handler_t *self, const frame_t *f);
} handler_ops_t;

struct handler {
    const handler_ops_t *ops;
    handler_t           *next;
    const char          *name;
};

/* ① 长度检查 */
static int len_handle(handler_t *h, const frame_t *f)
{
    if (f->len != 8) {
        printf("  [%s] 丢弃 0x%03X：长度 %d != 8\n", h->name, f->id, f->len);
        return 1;
    }
    printf("  [%s] 通过\n", h->name);
    return h->next ? h->next->ops->handle(h->next, f) : 0;
}

/* ② CRC：约定 crc = 高字节 ^ 低字节 ^ 长度 */
static int crc_handle(handler_t *h, const frame_t *f)
{
    unsigned char want = (unsigned char)(((f->id >> 8) & 0xFF) ^ (f->id & 0xFF) ^ f->len);
    if (f->crc != want) {
        printf("  [%s] 丢弃 0x%03X：crc=%02X，应为 %02X\n", h->name, f->id, f->crc, want);
        return 1;
    }
    printf("  [%s] 通过\n", h->name);
    return h->next ? h->next->ops->handle(h->next, f) : 0;
}

/* ③ 白名单：只收 0x100 / 0x2A0 */
static int white_handle(handler_t *h, const frame_t *f)
{
    if (!(f->id == 0x100 || f->id == 0x2A0)) {
        printf("  [%s] 丢弃 0x%03X：不在白名单\n", h->name, f->id);
        return 1;
    }
    printf("  [%s] 通过\n", h->name);
    return h->next ? h->next->ops->handle(h->next, f) : 0;
}

/* ④ 业务处理：链尾 */
static int app_handle(handler_t *h, const frame_t *f)
{
    printf("  [%s] 已消费 0x%03X\n", h->name, f->id);
    return 1;
}

static const handler_ops_t LEN_OPS   = { len_handle };
static const handler_ops_t CRC_OPS   = { crc_handle };
static const handler_ops_t WHITE_OPS = { white_handle };
static const handler_ops_t APP_OPS   = { app_handle };

int main(void)
{
    handler_t h_app   = { &APP_OPS,   NULL, "app"   };
    handler_t h_white = { &WHITE_OPS, &h_app, "white" };
    handler_t h_crc   = { &CRC_OPS,   &h_white, "crc" };
    handler_t h_len   = { &LEN_OPS,   &h_crc, "len" };

    static const frame_t frames[] = {
        { 0x100, 7, 0x09 },   /* 长度错 */
        { 0x100, 8, 0x00 },   /* crc 错 */
        { 0x500, 8, 0x0D },   /* 不在白名单 */
        { 0x100, 8, 0x09 }    /* 全部通过 */
    };

    for (unsigned i = 0; i < sizeof frames / sizeof frames[0]; i++) {
        printf("frame 0x%03X len=%d crc=%02X\n", frames[i].id, frames[i].len, frames[i].crc);
        h_len.ops->handle(&h_len, &frames[i]);
    }
    return 0;
}
C++chain.cpp
// 责任链模式 · C++ 实现：链式装配 + 每节点自己决定放行或终止
// 编译：g++ -std=c++17 -Wall -Wextra chain.cpp -o chain_cpp
#include <cstdio>
#include <string>

struct Frame { int id; int len; unsigned char crc; };

enum class Result { Pass, Stop };

class Handler {
public:
    explicit Handler(std::string name) : name_(std::move(name)) {}
    virtual ~Handler() = default;

    Handler& setNext(Handler& n) { next_ = &n; return n; }

    // 模板方法：先问自己，再决定要不要往后传
    Result run(const Frame& f) {
        Result r = check(f);
        if (r == Result::Stop) return Result::Stop;
        return next_ ? next_->run(f) : Result::Pass;
    }

protected:
    const std::string& name() const { return name_; }

private:
    virtual Result check(const Frame& f) = 0;
    std::string name_;
    Handler* next_ = nullptr;
};

class LenCheck final : public Handler {
public:
    LenCheck() : Handler("len") {}
protected:
    Result check(const Frame& f) override {
        if (f.len != 8) {
            std::printf("  [%s] 丢弃 0x%03X：长度 %d != 8\n", name().c_str(), f.id, f.len);
            return Result::Stop;
        }
        std::printf("  [%s] 通过\n", name().c_str());
        return Result::Pass;
    }
};

class CrcCheck final : public Handler {
public:
    CrcCheck() : Handler("crc") {}
protected:
    Result check(const Frame& f) override {
        unsigned char want = static_cast<unsigned char>(((f.id >> 8) & 0xFF) ^ (f.id & 0xFF) ^ f.len);
        if (f.crc != want) {
            std::printf("  [%s] 丢弃 0x%03X：crc=%02X，应为 %02X\n",
                        name().c_str(), f.id, f.crc, want);
            return Result::Stop;
        }
        std::printf("  [%s] 通过\n", name().c_str());
        return Result::Pass;
    }
};

class WhiteList final : public Handler {
public:
    WhiteList() : Handler("white") {}
protected:
    Result check(const Frame& f) override {
        if (!(f.id == 0x100 || f.id == 0x2A0)) {
            std::printf("  [%s] 丢弃 0x%03X：不在白名单\n", name().c_str(), f.id);
            return Result::Stop;
        }
        std::printf("  [%s] 通过\n", name().c_str());
        return Result::Pass;
    }
};

class AppSink final : public Handler {
public:
    AppSink() : Handler("app") {}
protected:
    Result check(const Frame& f) override {
        std::printf("  [%s] 已消费 0x%03X\n", name().c_str(), f.id);
        return Result::Stop;
    }
};

int main() {
    LenCheck len;
    CrcCheck crc;
    WhiteList white;
    AppSink app;

    len.setNext(crc).setNext(white).setNext(app);

    const Frame frames[] = {
        { 0x100, 7, 0x09 },
        { 0x100, 8, 0x00 },
        { 0x500, 8, 0x0D },
        { 0x100, 8, 0x09 }
    };
    for (const auto& f : frames) {
        std::printf("frame 0x%03X len=%d crc=%02X\n", f.id, f.len, f.crc);
        len.run(f);
    }
    return 0;
}

四帧各死在一节：长度、CRC、白名单——最后一帧才走完全链。被丢弃的帧，后面的节点一行日志都没有，这就是“终止”与“拒绝”的区别。

04发一帧，看它死在哪一节

点四个预设帧中的任意一个，报文会从链首逐节往下走。通过的节点变绿，丢它的节点变红，后面的节点保持灰——不再被调用。

05C 与 C++：谁负责“记得转 next”

角色C 版C++ 版差别意味着什么
节点接口handler_ops_t 一个函数指针Handler 基类 + 纯虚 check两者都是一个间接层，C 的更轻
转发每个 handle 末尾自己写基类 run() 统一转发C 版容易漏写；C++ 版结构上漏不了
终止return 1Result::StopC 的 int 没有语义约束，写 -1 也会被当“已处理”
装配{ &OPS, &next, "name" } 逐个初始化setNext() 链式链式写法要注意它返回的是 n，不是自己
节点生命周期都在 main 栈上，链比节点活得短谁申请谁释放，链只借指针，不拥有节点
加一节加一个函数 + 一张 ops 表 + 改装配加一个子类 + 改装配都只动装配，不动已有节点

06代价与坑

  • 递归转发会吃栈。C 版每个节点都调 next->handle，链有 40 节就有 40 层栈。嵌入式任务的栈常常只有几 KB，链一长就炸；改成 while 循环遍历更稳。
  • 链成环，报文永不停。装配时 app->next = len 这类手误，运行时表现为“某一帧卡死”。加一个装配期的长度/环路检查，或者把 next 设成只写一次。
  • 返回值只有两个合法值。C 版返回 -1 或 2 都算“非 0”，调用方当终止处理，日志却写着别的原因。用 enum 或至少断言范围。
  • 顺序既是性能也是安全。白名单放在 CRC 前，能少算非法 ID 的校验；但若白名单来自可篡改配置，先 CRC 再信 ID 更安全。顺序要写进设计文档，别让它只是“装配时随手一排”。
  • 节点可以改 frame。谁都能改，后面节点看到的就是改过的数据。约定“检查节点只读”，要改数据就在链尾单独做。
  • 链尾必须明确终止。业务节点返回 Pass，最后落到 nullptr 也只是“没人处理”，上层无法区分“消费了”和“丢了”。
  • C++ 的 setNext 返回引用。a.setNext(b).setNext(c) 实际是 a→b、b→c，不是给 a 连两次；读代码时容易看错，写注释或改成 void setNext 更清楚。
