状态模式STATE

状态机的核心不是“状态”这个词，而是一张显式的转移关系：在当前状态下，哪些事件合法、合法时去哪、非法时怎么办。C 版把这张关系写成 5×6 的表，C++ 版把它拆进五个状态类；两边的行为和拒绝路径完全一致。

30 格转移表 · 12 条合法 非法转移统一拒绝 C++ 状态类 · 无动态分配

01问题：switch 里再套 if，非法组合没人管

常见的写法是：switch (state) 里对每个事件写一段 if，命中就改 state。它能跑，但三个问题会慢慢长出来：

  • 非法组合没人拦。没写的组合直接落到 default 或者什么也不做，故障下按错键也不会报；
  • 改完状态还继续往下跑。一段处理里改了 state，后面几行仍按旧状态的假设执行，容易出现一次事件跳两步；
  • 动作和转移混在一起。“进 RUN 要开泵、进 FAULT 要关输出”塞在分支里，想单测转移路径就得把硬件全模拟一遍。

状态模式把转移关系先抽出来，动作再挂到“进入某状态”或“转移成功”上。这样表能被静态检查、能被穷举测试，代码里也不会再有“忘了写某个组合”的洞。

03两份实现

C 版把 30 个格子摊成一张常量二维数组，-1 表示“这个状态下不接受这个事件”；C++ 版把每一行搬进一个状态类，返回 nullptr 表示拒绝。分派形式不同，拒绝点完全一致。

Cstate.c
/* 状态模式 · C 实现：ECU 电源/运行状态机，转移写成表
 * 编译：gcc -std=c11 -Wall -Wextra state.c -o state
 * 5 个状态 × 6 个事件 = 30 个格子，非法格子统一返回 -1。
 */
#include <stdio.h>

enum { ST_OFF, ST_ACC, ST_ON, ST_RUN, ST_FAULT, ST_N };
enum { EV_ACC, EV_ON, EV_CRANK, EV_OFF, EV_OVER, EV_CLEAR, EV_N };

static const char *const ST_NAME[] = { "OFF", "ACC", "ON", "RUN", "FAULT" };
static const char *const EV_NAME[] = { "ACC", "ON", "CRANK", "OFF", "OVERTEMP", "CLEAR" };

/* 行 = 当前状态，列 = 事件；-1 表示该状态下这个事件非法 */
static const int TRANS[ST_N][EV_N] = {
    /* OFF   */ { ST_ACC, -1,     -1,      -1,     -1,       -1     },
    /* ACC   */ { -1,     ST_ON,  -1,      ST_OFF, ST_FAULT, -1     },
    /* ON    */ { ST_ACC, -1,     ST_RUN,  ST_OFF, ST_FAULT, -1     },
    /* RUN   */ { -1,     -1,     -1,      ST_OFF, ST_FAULT, -1     },
    /* FAULT */ { -1,     -1,     -1,      ST_OFF, -1,       ST_OFF }
};

typedef struct { int state; } ecu_t;

static int ecu_inject(ecu_t *e, int ev)
{
    int next = TRANS[e->state][ev];
    if (next < 0) {
        printf("[ECU] %-5s + %-8s -> 拒绝（该状态下事件非法）\n",
               ST_NAME[e->state], EV_NAME[ev]);
        return -1;
    }
    printf("[ECU] %-5s + %-8s -> %s\n", ST_NAME[e->state], EV_NAME[ev], ST_NAME[next]);
    e->state = next;
    return 0;
}

int main(void)
{
    ecu_t e = { ST_OFF };
    static const int seq[] = { EV_ACC, EV_ON, EV_CRANK, EV_OVER, EV_CRANK, EV_CLEAR, EV_CRANK };
    static const char *const why[] = {
        "钥匙拧到 ACC", "再拧到 ON（点火档）", "起动机拖动，发动机运转", "冷却液超温",
        "已进故障，再打起动机", "故障清除", "已断电，再打起动机"
    };

    for (unsigned i = 0; i < sizeof seq / sizeof seq[0]; i++) {
        printf("--- %s ---\n", why[i]);
        ecu_inject(&e, seq[i]);
    }
    printf("最终状态：%s\n", ST_NAME[e.state]);

    int legal = 0;
    for (int s = 0; s < ST_N; s++)
        for (int k = 0; k < EV_N; k++)
            if (TRANS[s][k] >= 0) legal++;
    printf("转移表 %d 个格子，其中合法转移 %d 个\n", ST_N * EV_N, legal);
    return 0;
}
C++state.cpp
// 状态模式 · C++ 实现：每个状态一个类，由状态自己决定下一步去哪
// 编译：g++ -std=c++17 -Wall -Wextra state.cpp -o state_cpp
#include <cstdio>

enum class Event { Acc, On, Crank, Off, OverTemp, Clear };

const char* evName(Event e) {
    switch (e) {
        case Event::Acc: return "ACC";
        case Event::On: return "ON";
        case Event::Crank: return "CRANK";
        case Event::Off: return "OFF";
        case Event::OverTemp: return "OVERTEMP";
        case Event::Clear: return "CLEAR";
    }
    return "?";
}

class Ecu;
class State {
public:
    virtual ~State() = default;
    virtual const char* name() const = 0;
    virtual State* on(Event e, Ecu& ctx) = 0;   // 返回 nullptr 表示拒绝
};

class Off : public State {
public:
    const char* name() const override { return "OFF"; }
    State* on(Event e, Ecu& ctx) override;
};
class Acc : public State {
public:
    const char* name() const override { return "ACC"; }
    State* on(Event e, Ecu& ctx) override;
};
class On : public State {
public:
    const char* name() const override { return "ON"; }
    State* on(Event e, Ecu& ctx) override;
};
class Run : public State {
public:
    const char* name() const override { return "RUN"; }
    State* on(Event e, Ecu& ctx) override;
};
class Fault : public State {
public:
    const char* name() const override { return "FAULT"; }
    State* on(Event e, Ecu& ctx) override;
};

// 函数内静态对象：避免静态初始化顺序问题
static Off&   stOff()   { static Off s;   return s; }
static Acc&   stAcc()   { static Acc s;   return s; }
static On&    stOn()    { static On s;    return s; }
static Run&   stRun()   { static Run s;   return s; }
static Fault& stFault() { static Fault s; return s; }

class Ecu {
public:
    explicit Ecu(State* s) : st_(s) {}
    void inject(Event e) {
        State* next = st_->on(e, *this);
        if (!next) {
            std::printf("[ECU] %-5s + %-8s -> 拒绝（该状态下事件非法）\n",
                        st_->name(), evName(e));
            return;
        }
        std::printf("[ECU] %-5s + %-8s -> %s\n", st_->name(), evName(e), next->name());
        st_ = next;
    }
    const char* name() const { return st_->name(); }
private:
    State* st_;
};

State* Off::on(Event e, Ecu&) {
    if (e == Event::Acc) return &stAcc();
    return nullptr;
}
State* Acc::on(Event e, Ecu&) {
    switch (e) {
        case Event::On: return &stOn();
        case Event::Off: return &stOff();
        case Event::OverTemp: return &stFault();
        default: return nullptr;
    }
}
State* On::on(Event e, Ecu&) {
    switch (e) {
        case Event::Crank: return &stRun();
        case Event::Off: return &stOff();
        case Event::Acc: return &stAcc();
        case Event::OverTemp: return &stFault();
        default: return nullptr;
    }
}
State* Run::on(Event e, Ecu&) {
    switch (e) {
        case Event::Off: return &stOff();
        case Event::OverTemp: return &stFault();
        default: return nullptr;
    }
}
State* Fault::on(Event e, Ecu&) {
    switch (e) {
        case Event::Off: return &stOff();
        case Event::Clear: return &stOff();
        default: return nullptr;
    }
}

int main() {
    Ecu e(&stOff());
    const Event seq[] = { Event::Acc, Event::On, Event::Crank, Event::OverTemp,
                          Event::Crank, Event::Clear, Event::Crank };
    const char* why[] = {
        "钥匙拧到 ACC", "再拧到 ON（点火档）", "起动机拖动，发动机运转", "冷却液超温",
        "已进故障，再打起动机", "故障清除", "已断电，再打起动机"
    };
    for (int i = 0; i < 7; ++i) {
        std::printf("--- %s ---\n", why[i]);
        e.inject(seq[i]);
    }
    std::printf("最终状态：%s\n", e.name());
    return 0;
}

05C 与 C++：同一张表，两种摆放

角色C 版C++ 版差别意味着什么
状态集合enum ST_* 整数class Off/Acc/On/Run/Fault整数便宜但可被任意强转；类型安全要多一层检查
转移定义TRANS[状态][事件] 常量表各状态类的 on()表能一眼看全、能穷举；类把复杂度摊到文件各处
非法判定-1nullptr两者都是“显式拒绝”，不靠 default 兜底
当前状态int stateState* st_C++ 多一次间接寻址；C 多一次二维下标
生命周期无函数内静态单例没有 new，就不存在“忘了 delete 状态对象”
加一个状态加一行、改 enum、改所有行加一个类 + 挂进相关 on()表改一处但容易漏列；类改动分散但编译器会提醒未覆盖
动作副作用转移成功后统一分发可在 onEnter 钩子里做别塞进转移表，表要保持“纯数据”

06代价与坑

  • 表越界比逻辑错更难查。TRANS[state][ev] 里 state 或 ev 若来自外部（CAN 信号、配置），越界就是未定义行为。入口处先做范围校验，别指望 -1 兜住。
  • 拒绝要能被上层看见。return -1 但调用方不看返回值，等于没做保护。至少让拒绝有日志和计数，诊断时才知道“用户按了但车没反应”。
  • 一次事件只走一步。在 on() 里递归 inject 会连跳多格，日志和外部观察者都对不上。需要连跳就显式发两次事件。
  • CLEAR 不是复位。故障清除只回 OFF，不代表清故障码、不代表重启。这类语义要在文档里写死，否则后期一定有人拿 CLEAR 当 reboot 用。
  • 状态类别持有可变数据。把状态做成静态单例时，它只能是“规则”，不能存当前车速、当前档位；那些属于上下文 Ecu。
  • 30 格维护成本。状态和事件一多，矩阵会变成 8×12、10×15；这时就该用脚本从 YAML/CSV 生成表，而不是手写。
