    备忘录模式MEMENTO

    现场调参最怕“越调越远”：增益、偏移、模式改了十几轮，想回到最初那版却没有记录。备忘录把“当前状态”存成一份不可变快照，需要时整体恢复；原对象不用暴露内部字段给外部。

    改前先存快照不可变恢复是整体替换

    01问题：调参调坏了，没有退路

    没有快照时，回退靠“记得原来的值”：

    ecu.gain = 30;   /* 原来是多少？ */
ecu.offset = -8; /* 谁记了？ */
    • 回退不完整。只记住 gain，忘了 mode，恢复后状态是混合的。
    • 快照被污染。把 calib_t* 直接存起来，后续改原对象，快照跟着变。
    • 外部乱改。为了让调用方能恢复，把内部字段全暴露，封装名存实亡。

    备忘录把快照做成独立值：保存时拷贝，恢复时整体替换；快照只给原对象读写，外部只能“拿着”，不能“改里面的字段”。

    02结构：快照与恢复分开

    原对象、快照、保管者三方：点方框看谁有权读写快照。

    标定快照：保存与恢复分开memento
    职责点方框：原对象写快照，保管者只搬运。

    03两份实现

    C 版快照是结构体值；C++ 版把 Memento 的字段设为私有，只对 Ecu 开放。

    Cmemento.c
    /* 备忘录：改标定之前先存一份，出事能原样退回 */
#include <stdio.h>

typedef struct {
    int gain;
    int offset;
    int mode;
} calib_t;

typedef struct { calib_t snap; } memento_t;

static memento_t save(const calib_t *c)
{
    memento_t m = { *c };
    return m;
}
static void restore(calib_t *c, const memento_t *m)
{
    *c = m->snap;
}

static void dump(const char *tag, const calib_t *c)
{
    printf("%-9s gain=%d offset=%d mode=%d\n", tag, c->gain, c->offset, c->mode);
}

int main(void)
{
    calib_t ecu = { 12, 0, 1 };
    dump("before", &ecu);

    memento_t m = save(&ecu);

    ecu.gain = 30;          /* 现场调参，调坏了 */
    ecu.offset = -8;
    ecu.mode = 3;
    dump("tuned", &ecu);

    restore(&ecu, &m);
    dump("restored", &ecu);
    printf("快照是值拷贝：之后改 ecu 不会污染 memento\n");
    return 0;
}
    RUN./memento.exe
    before    gain=12 offset=0 mode=1
tuned     gain=30 offset=-8 mode=3
restored  gain=12 offset=0 mode=1

    C++memento.cpp
    // 备忘录 · C++：Memento 只暴露给 Originator，外部拿不到内部字段
// 编译：g++ -std=c++17 -Wall -Wextra memento.cpp -o memento_cpp
#include <cstdio>

struct Calib { int gain; int offset; int mode; };

class Memento {
public:
    explicit Memento(const Calib &c) : c_(c) {}
private:
    friend class Ecu;
    Calib c_;
};

class Ecu {
public:
    void set(int g, int o, int m) { c_ = Calib{g, o, m}; }
    Memento save() const { return Memento(c_); }
    void restore(const Memento &m) { c_ = m.c_; }
    void dump(const char *tag) const
    {
        std::printf("%-9s gain=%d offset=%d mode=%d\n", tag, c_.gain, c_.offset, c_.mode);
    }
private:
    Calib c_{};
};

int main()
{
    Ecu ecu;
    ecu.set(12, 0, 1);
    ecu.dump("before");
    Memento m = ecu.save();

    ecu.set(30, -8, 3);
    ecu.dump("tuned");

    ecu.restore(m);
    ecu.dump("restored");
    std::printf("快照是值拷贝：之后改 ecu 不会污染 memento\n");
    return 0;
}
    RUN./memento_cpp.exe
    before    gain=12 offset=0 mode=1
tuned     gain=30 offset=-8 mode=3
restored  gain=12 offset=0 mode=1

    05C 与 C++：快照的可见性

    角色C 版C++ 版差别意味着什么
    快照memento_t 内嵌 calib_tMemento 私有字段C 靠约定不碰；C++ 用访问控制
    保存save(&ecu) 值拷贝ecu.save()都要求深拷贝
    恢复restore(&ecu, &m)ecu.restore(m)整体替换，不逐字段挑
    可见性结构体公开，谁都能改仅 friend Ecu 可读写C++ 能把“不可变”落到编译期
    多份历史自己开数组存vector<Memento>要撤销栈就多存几份

    06代价与坑

    • 快照必须是深拷贝。配置里有字符串或指针时，浅拷贝只复制地址，原对象一改快照就变。
    • 别把快照当历史日志。备忘录通常只保一份；需要多级撤销要显式做栈，并限制深度。
    • 频繁保存有成本。每改一个字段存一次，内存和拷贝开销都上来；按“一次调参会话”存一次更合理。
    • 恢复不等于回滚外部副作用。标定恢复了，已经写进 NVRAM 的值、已经发出的报文不会自动撤回。
    • 快照别跨线程共享。一个快照被两个线程同时恢复，读到的可能是半更新状态；要么加锁，要么每线程各存一份。
