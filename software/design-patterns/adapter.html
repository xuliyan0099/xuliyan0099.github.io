适配器模式ADAPTER

新代码只想要一件事：read() 给我温度，单位是摄氏度。可手上的两个旧驱动一个返回 ×100 的整数还要求先 init()，另一个只给 ADC 计数、根本没有湿度。适配器就是那个“翻译层”：把两边的差异吸收在自己身上，业务代码一行都不用改。

两个旧驱动 · 一个接口 单位换算在适配器 能力缺失要如实报

01问题：差异会一路渗到业务层

两个传感器的驱动 API 完全不一样：

  • SHT3x 的库返回 int t_x100，还要求先调 init() 打开器件；
  • NTC 只有一个 ADC 计数，温度得自己按分压表算，湿度则压根没有这个量。

如果让业务直接调这两个驱动，业务代码里就会出现“if 是 SHT3x 就除以 100，if 是 NTC 就套公式”的分支。每多一个型号、每换一次单位，都要改业务；同时业务也被迫知道“谁需要 init、谁没有湿度”。

适配器把这段差异收进一个类：对外仍然是“读温度、读湿度”，对内负责换算、补协议、把“不支持”翻译成明确错误码。业务只依赖目标接口，换传感器等于换一个适配器对象。

02结构：目标接口在前，适配器在中间

上层只看 ISensor；两个适配器各自包住一个旧驱动，把它的返回值翻译成统一读数。点方框看每一层负责什么。

对象适配器（包住旧驱动）target / adapter / adaptee
职责点方框：为什么“不支持湿度”也要走接口，而不是业务里写 if。

03两份实现

C 版的目标是“一个 ops 表 + 一个 impl 指针”，适配器结构体把 sensor_t base 放在第一位再强转回来；C++ 版是标准的对象适配器：适配器内部持有一个旧驱动对象，对外实现 read()。

Cadapter.c
/* 适配器模式 · C 实现：两种温湿度传感器，上层只认一套接口
 * 编译：gcc -std=c11 -Wall -Wextra adapter.c -o adapter
 */
#include <stdio.h>

/* ---------- 目标接口：上层代码只认这三个成员 ---------- */
typedef struct sensor sensor_t;
typedef struct {
    int (*read_temp)(sensor_t *self, float *out);
    int (*read_hum)(sensor_t *self, float *out);
} sensor_ops_t;

struct sensor { const sensor_ops_t *ops; void *impl; };

/* ---------- 被适配者 A：SHT3x 厂商库，返回 ×100 的整数 ---------- */
typedef struct { int t_x100; int h_x100; } sht3x_t;

static int sht3x_legacy_read(sht3x_t *d, int *t_x100, int *h_x100)
{
    if (!d) return -1;
    *t_x100 = d->t_x100;
    *h_x100 = d->h_x100;
    return 0;
}

typedef struct { sensor_t base; sht3x_t hw; } sht_adapter_t;

static int sht_read_temp(sensor_t *self, float *out)
{
    sht_adapter_t *a = (sht_adapter_t *)self;     /* 适配器知道两边 */
    int t, h;
    if (sht3x_legacy_read(&a->hw, &t, &h) != 0) return -1;
    *out = t / 100.0f;
    return 0;
}

static int sht_read_hum(sensor_t *self, float *out)
{
    sht_adapter_t *a = (sht_adapter_t *)self;
    int t, h;
    if (sht3x_legacy_read(&a->hw, &t, &h) != 0) return -1;
    *out = h / 100.0f;
    return 0;
}

static const sensor_ops_t SHT_OPS = { sht_read_temp, sht_read_hum };

/* ---------- 被适配者 B：NTC + ADC，只能测温，湿度不支持 ---------- */
typedef struct { int adc; } ntc_hw_t;

static int ntc_read_adc(ntc_hw_t *d) { return d->adc; }

typedef struct { sensor_t base; ntc_hw_t hw; } ntc_adapter_t;

static int ntc_read_temp(sensor_t *self, float *out)
{
    ntc_adapter_t *a = (ntc_adapter_t *)self;
    int adc = ntc_read_adc(&a->hw);
    *out = 25.0f + (adc - 2048) * 0.05f;          /* 分压表的线性近似 */
    return 0;
}

static int ntc_read_hum(sensor_t *self, float *out)
{
    (void)self; (void)out;
    return -2;                                   /* 能力缺失要如实报错 */
}

static const sensor_ops_t NTC_OPS = { ntc_read_temp, ntc_read_hum };

/* ---------- 上层：同一段代码，两个实现 ---------- */
static void report(const char *tag, sensor_t *s)
{
    float t = 0, h = 0;
    printf("%s: ", tag);
    if (s->ops->read_temp(s, &t) == 0) printf("temp=%.1fC ", t);
    else                               printf("temp=<err> ");
    int rc = s->ops->read_hum(s, &h);
    if (rc == 0)      printf("hum=%.1f%%\n", h);
    else if (rc == -2) printf("hum=<不支持>\n");
    else               printf("hum=<err>\n");
}

int main(void)
{
    sht_adapter_t a = { { &SHT_OPS, NULL }, { 2360, 4850 } };
    ntc_adapter_t b = { { &NTC_OPS, NULL }, { 2100 } };
    a.base.impl = &a.hw;
    b.base.impl = &b.hw;

    report("SHT3x", &a.base);
    report("NTC  ", &b.base);
    return 0;
}
C++adapter.cpp
// 适配器模式 · C++ 实现：对象适配器（包住旧驱动）与接口统一
// 编译：g++ -std=c++17 -Wall -Wextra adapter.cpp -o adapter_cpp
#include <cstdio>

// 目标接口：新代码只依赖它
struct Reading { bool ok; float temp_c; float hum_pct; };

class ISensor {
public:
    virtual ~ISensor() = default;
    virtual Reading read() = 0;
};

// 被适配者 A：厂商库，整数 + 缩放，且要求先 init
class Sht3xDriver {
public:
    bool init() { ready_ = true; return true; }
    bool readRaw(int& t_x100, int& h_x100) {
        if (!ready_) return false;
        t_x100 = 2360;
        h_x100 = 4850;
        return true;
    }
private:
    bool ready_ = false;
};

class Sht3xAdapter final : public ISensor {
public:
    Sht3xAdapter() { drv_.init(); }
    Reading read() override {
        int t = 0, h = 0;
        if (!drv_.readRaw(t, h)) return {};
        return { true, t / 100.0f, h / 100.0f };
    }
private:
    Sht3xDriver drv_;
};

// 被适配者 B：ADC 原始计数，只能测温
class NtcAdc {
public:
    int rawCount() const { return 2100; }
};

class NtcAdapter final : public ISensor {
public:
    Reading read() override {
        float t = 25.0f + (adc_.rawCount() - 2048) * 0.05f;
        return { true, t, -1.0f };     // -1 表示湿度不可用
    }
private:
    NtcAdc adc_;
};

void report(const char* tag, ISensor& s) {
    Reading r = s.read();
    std::printf("%s: ", tag);
    if (r.ok) std::printf("temp=%.1fC ", r.temp_c);
    else      std::printf("temp=<err> ");
    if (r.ok && r.hum_pct >= 0) std::printf("hum=%.1f%%\n", r.hum_pct);
    else                        std::printf("hum=<不支持>\n");
}

int main() {
    Sht3xAdapter a;
    NtcAdapter b;
    report("SHT3x", a);
    report("NTC  ", b);
    return 0;
}
RUN./adapter.exe
SHT3x: temp=23.6C hum=48.5%
NTC  : temp=27.6C hum=<不支持>
RUN./adapter_cpp.exe
SHT3x: temp=23.6C hum=48.5%
NTC  : temp=27.6C hum=<不支持>

两行完全一致。NTC 的 27.6 来自 25 + (2100-2048)×0.05；湿度那一格没有变成 0.0，而是明确写“不支持”——这是适配器最容易被写错的地方。

04改原始值，看适配器怎么翻译

拖动滑块改变驱动吐出的原始值。SHT3x 走 ×100 换算，NTC 走分压表近似；NTC 的湿度永远走“不支持”这条分支，而不是编一个数出来。

05C 与 C++：同一层翻译，两种代价

角色C 版C++ 版差别意味着什么
目标接口sensor_ops_t 两个函数指针ISensor::read()C 接口是数据，能被拷贝、能被替换
适配器sensor_t base + hw类内持有 Sht3xDriver都是“包住”，C 的包法靠内存布局
还原具体类型(sht_adapter_t *)self虚分派直达具体类C 版强转无检查，布局错就是 UB
只读一次拿两个值两个方法各调一次 legacy_read一次 read() 返回 ReadingC 版多读一次 I2C；C++ 版一次读完
错误表达-1 失败、-2 不支持ok=false、hum_pct=-1用“魔法负数”区分两种错误，得写清注释
开销无分配无堆分配（驱动内嵌）两种都适合资源紧张的板子

06代价与坑

  • 能力缺失不能编数。NTC 没有湿度，返回 0.0 会被上层当成“湿度 0%”，触发加湿器满功率。必须用独立错误码，并且上层要能区分“不支持”和“读失败”。
  • 适配器里不要加业务。滤波、标定、单位转换之外的逻辑（比如“温度超 60 就报警”）不属于翻译层。塞进去之后，第二个适配器也得抄一遍，差异又回来了。
  • C 版布局是隐式契约。base 必须在结构体第一位；用 _Static_assert(offsetof(sht_adapter_t, base) == 0) 把它钉死。否则某次“优化对齐”就会让所有强转失效。
  • 状态不能放静态变量。两个 SHT3x 实例共用一个 ops 表是正常的，但如果适配器把 t_x100 存在文件级静态变量里，第二个传感器会把第一个的数据覆盖。状态永远跟着实例走。
  • init 失败要能上报。C++ 版构造函数里 drv_.init() 的返回值被丢掉了；工程上要么在构造后单独 begin()，要么把失败状态存起来让第一次 read() 返回 ok=false。
  • 适配器是单向的。它让旧驱动满足新接口，不会反过来让新代码去迁就旧 API。两代 API 并存时，通常各配一个适配器，而不是做一个“万能转换”。
  • 整数除法。C 版 t / 100.0f 是对的，写成 t / 100 会把 23.6 变成 23 甚至 0（负数截断方向还不同）。这类错误在编译期没有任何提示。
